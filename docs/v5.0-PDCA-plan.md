```python
# core/rdf_migration.py
import sqlite3, shutil, json
from datetime import datetime
from pathlib import Path
from rdflib import Graph, Namespace, URIRef, Literal, RDF, RDFS, OWL
from rdflib.namespace import FOAF, XSD, DCTERMS

class RDFMigrationService:
    def __init__(self):
        self.db_path   = Path("data/oral_record.db")
        self.rdf_dir   = Path("data/rdf")
        self.backup_dir= Path("data/backups")
        self.rdf_dir.mkdir(parents=True, exist_ok=True)
        self.backup_dir.mkdir(parents=True, exist_ok=True)
        self.state_path = Path("data/migration_state.json")

    # ── 상태 관리 ────────────────────────────────────────────
    def get_status(self) -> dict:
        conn = sqlite3.connect(self.db_path)
        try:
            ontology_count = conn.execute(
                "SELECT COUNT(*) FROM ontologies").fetchone()[0]
            triple_count   = conn.execute(
                "SELECT COUNT(*) FROM triples").fetchone()[0]
            record_count   = conn.execute(
                "SELECT COUNT(*) FROM records").fetchone()[0]
        except:
            ontology_count = triple_count = record_count = 0
        finally:
            conn.close()

        rdf_exists  = (self.rdf_dir / "oral-history.ttl").exists()
        backup_list = list(self.backup_dir.glob("*.db"))
        state       = self._load_state()

        return {
            "sqlite": {
                "ontology_count": ontology_count,
                "triple_count":   triple_count,
                "record_count":   record_count,
            },
            "rdf_converted": rdf_exists,
            "rdf_confirmed": state.get("confirmed", False),
            "backup_files":  [f.name for f in backup_list],
            "last_migration": state.get("last_migration"),
        }

    # ── DB 분석 ──────────────────────────────────────────────
    def analyze_db(self) -> dict:
        conn   = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute("SELECT name FROM sqlite_master WHERE type='table'")
        tables = [row[0] for row in cursor.fetchall()]
        result = {}
        for table in tables:
            cursor.execute(f"SELECT COUNT(*) FROM {table}")
            count = cursor.fetchone()[0]
            cursor.execute(f"PRAGMA table_info({table})")
            cols = [col[1] for col in cursor.fetchall()]
            result[table] = {"count": count, "columns": cols}
        conn.close()
        return {"tables": result}

    # ── 백업 ─────────────────────────────────────────────────
    def backup(self) -> dict:
        ts  = datetime.now().strftime("%Y%m%d_%H%M%S")
        dst = self.backup_dir / f"oral_record_v4_backup_{ts}.db"
        shutil.copy2(self.db_path, dst)
        return {"backup_file": str(dst), "size_kb": dst.stat().st_size // 1024}

    # ── 마이그레이션 실행 ─────────────────────────────────────
    def run(self, namespace_uri: str, formats: list) -> dict:
        ORA  = Namespace(namespace_uri)
        CRM  = Namespace("http://www.cidoc-crm.org/cidoc-crm/")
        RICO = Namespace("https://www.ica.org/standards/RiC/ontology#")
        SCH  = Namespace("https://schema.org/")

        g = Graph()
        g.bind("ora",     ORA)
        g.bind("crm",     CRM)
        g.bind("rico",    RICO)
        g.bind("foaf",    FOAF)
        g.bind("schema",  SCH)
        g.bind("dcterms", DCTERMS)

        # 온톨로지 선언
        onto_uri = URIRef(namespace_uri.rstrip("#/"))
        g.add((onto_uri, RDF.type, OWL.Ontology))
        g.add((onto_uri, RDFS.label,
               Literal("한국 구술기록 온톨로지", lang="ko")))
        g.add((onto_uri, OWL.versionInfo, Literal("1.0.0")))

        conn = sqlite3.connect(self.db_path)

        # 온톨로지 클래스 + 매핑 변환
        try:
            for row in conn.execute(
                "SELECT id, name, reference_ontology, mapped_class, status FROM ontologies"
            ):
                oid, name, ref_onto, mapped, status = row
                safe_name = self._uri_safe(name)
                cls_uri   = ORA[safe_name]
                g.add((cls_uri, RDF.type,    OWL.Class))
                g.add((cls_uri, RDFS.label,  Literal(name, lang="ko")))
                if mapped:
                    g.add((cls_uri, RDFS.subClassOf, URIRef(mapped)))
        except Exception as e:
            pass

        # 트리플 변환
        try:
            for row in conn.execute(
                "SELECT subject, predicate, object FROM triples"
            ):
                s, p, o = row
                s_uri = ORA[self._uri_safe(s)]
                p_uri = ORA[self._uri_safe(p)]
                g.add((s_uri, p_uri, Literal(o)))
        except Exception as e:
            pass

        conn.close()

        # 임시 디렉토리에 저장 (확정 전)
        tmp_dir = self.rdf_dir / "tmp"
        tmp_dir.mkdir(exist_ok=True)

        format_map = {
            "turtle":  ("oral-history.ttl",    "turtle"),
            "json-ld": ("oral-history.jsonld",  "json-ld"),
            "xml":     ("oral-history.owl",     "xml"),
        }
        saved = []
        for fmt in formats:
            if fmt in format_map:
                fname, rfmt = format_map[fmt]
                out = tmp_dir / fname
                g.serialize(str(out), format=rfmt)
                saved.append(str(out))

        self._save_state({"last_migration": datetime.now().isoformat(),
                          "triple_count": len(g), "confirmed": False})
        return {"triple_count": len(g), "saved_files": saved}

    # ── 미리보기 ──────────────────────────────────────────────
    def preview(self, limit: int = 50) -> dict:
        tmp_ttl = self.rdf_dir / "tmp" / "oral-history.ttl"
        if not tmp_ttl.exists():
            return {"error": "마이그레이션을 먼저 실행하세요"}
        g = Graph()
        g.parse(str(tmp_ttl), format="turtle")
        triples = []
        for i, (s, p, o) in enumerate(g):
            if i >= limit:
                break
            triples.append({"s": str(s), "p": str(p), "o": str(o)})
        return {"total": len(g), "preview_count": len(triples),
                "triples": triples,
                "turtle_sample": g.serialize(format="turtle")[:2000]}

    # ── 확정 ─────────────────────────────────────────────────
    def confirm(self) -> dict:
        tmp_dir  = self.rdf_dir / "tmp"
        final_dir= self.rdf_dir / "v1.0"
        final_dir.mkdir(exist_ok=True)
        for f in tmp_dir.glob("*"):
            shutil.copy2(f, final_dir / f.name)
            # latest 심링크 역할로 rdf/ 루트에도 복사
            shutil.copy2(f, self.rdf_dir / f.name)
        self._save_state({"confirmed": True,
                          "confirmed_at": datetime.now().isoformat()})
        return {"status": "confirmed", "path": str(final_dir)}

    # ── 롤백 ─────────────────────────────────────────────────
    def rollback(self) -> dict:
        backups = sorted(self.backup_dir.glob("*.db"), reverse=True)
        if not backups:
            return {"error": "백업 파일이 없습니다"}
        latest = backups[0]
        shutil.copy2(latest, self.db_path)
        # RDF 임시 파일 삭제
        tmp_dir = self.rdf_dir / "tmp"
        if tmp_dir.exists():
            shutil.rmtree(tmp_dir)
        return {"status": "rolled_back", "restored_from": latest.name}

    # ── 내부 유틸 ─────────────────────────────────────────────
    def _uri_safe(self, text: str) -> str:
        import re
        return re.sub(r'[^\w]', '_', str(text))

    def _load_state(self) -> dict:
        if self.state_path.exists():
            return json.loads(self.state_path.read_text(encoding="utf-8"))
        return {}

    def _save_state(self, data: dict):
        current = self._load_state()
        current.update(data)
        self.state_path.write_text(
            json.dumps(current, ensure_ascii=False, indent=2), encoding="utf-8")
```

# api/router_rdf_ontology.py
from fastapi import APIRouter, HTTPException
from pydantic import BaseModel
from core.rdf_store import get_rdf_store
from rdflib import URIRef, Literal, RDF, RDFS, OWL
from rdflib.namespace import FOAF

router = APIRouter(prefix="/rdf/ontology", tags=["RDF 온톨로지 편집"])

class ClassModel(BaseModel):
    uri: str
    label_ko: str
    label_en: str = ""
    parent_class: str = ""
    description: str = ""

@router.get("/classes")
def list_classes():
    store = get_rdf_store()
    results = store.query("""
        SELECT ?cls ?label_ko ?label_en ?parent WHERE {
            ?cls a owl:Class .
            OPTIONAL { ?cls rdfs:label ?label_ko
                       FILTER(lang(?label_ko) = "ko") }
            OPTIONAL { ?cls rdfs:label ?label_en
                       FILTER(lang(?label_en) = "en") }
            OPTIONAL { ?cls rdfs:subClassOf ?parent }
        }
    """)
    return [{"uri": str(r.cls),
             "label_ko": str(r.label_ko) if r.label_ko else "",
             "label_en": str(r.label_en) if r.label_en else "",
             "parent":   str(r.parent)   if r.parent   else ""}
            for r in results]

@router.put("/classes/{class_local}")
def update_class(class_local: str, data: ClassModel):
    store = get_rdf_store()
    uri   = URIRef(data.uri)
    # 기존 레이블 제거 후 재추가
    store.graph.remove((uri, RDFS.label, None))
    store.graph.remove((uri, RDFS.subClassOf, None))
    store.graph.add((uri, RDFS.label, Literal(data.label_ko, lang="ko")))
    if data.label_en:
        store.graph.add((uri, RDFS.label, Literal(data.label_en, lang="en")))
    if data.parent_class:
        store.graph.add((uri, RDFS.subClassOf, URIRef(data.parent_class)))
    store._save()
    return {"status": "updated"}

@router.get("/turtle")

    return {"turtle": store.serialize(format="turtle")}

# core/rdf_validator.py
from rdflib import Graph, RDF, RDFS, OWL, URIRef
from pathlib import Path
import json
from datetime import datetime

class RDFValidator:
    def __init__(self):
        self.rdf_path  = Path("data/rdf/oral-history.ttl")
        self.result_path = Path("data/validation_results.json")

    def run_all(self) -> dict:
        if not self.rdf_path.exists():
            return {"error": "RDF 파일 없음. 마이그레이션을 먼저 실행하세요."}
        g = Graph()
        g.parse(str(self.rdf_path), format="turtle")

        results = {
            "timestamp": datetime.now().isoformat(),
            "triple_count": len(g),
            "checks": {}
        }
        results["checks"]["structure"] = self._check_structure(g)
        results["checks"]["labels"]    = self._check_labels(g)
        results["checks"]["mappings"]  = self._check_mappings(g)
        results["checks"]["duplicates"]= self._check_duplicates(g)
        results["checks"]["consistency"]= self._check_consistency()

        # 준비도 계산
        checks   = results["checks"]
        passed   = sum(1 for c in checks.values() if c["status"] == "pass")
        total    = len(checks)
        warnings = sum(len(c.get("warnings", [])) for c in checks.values())
        errors   = sum(len(c.get("errors",   [])) for c in checks.values())

        results["summary"] = {
            "passed": passed, "total": total,
            "warnings": warnings, "errors": errors,
            "readiness": max(0, 100 - errors * 20 - warnings * 5)
        }
        # 결과 저장
        self.result_path.write_text(
            json.dumps(results, ensure_ascii=False, indent=2), encoding="utf-8")
        return results

    def _check_structure(self, g) -> dict:
        classes = list(g.subjects(RDF.type, OWL.Class))
        errors  = []
        for cls in classes:
            if not isinstance(cls, URIRef):
                errors.append(f"익명 클래스 발견: {cls}")
        return {"status": "pass" if not errors else "error",
                "class_count": len(classes), "errors": errors}

    def _check_labels(self, g) -> dict:
        classes   = list(g.subjects(RDF.type, OWL.Class))
        missing   = []
        for cls in classes:
            labels = list(g.objects(cls, RDFS.label))
            ko = [l for l in labels if getattr(l, 'language', '') == 'ko']
            if not ko:
                missing.append(str(cls))
        return {
            "status":   "pass" if not missing else "warning",
            "total":    len(classes),
            "with_ko":  len(classes) - len(missing),
            "warnings": [f"한국어 레이블 없음: {u}" for u in missing]
        }

    def _check_mappings(self, g) -> dict:
        classes  = list(g.subjects(RDF.type, OWL.Class))
        unmapped = []
        for cls in classes:
            if not list(g.objects(cls, RDFS.subClassOf)):
                unmapped.append(str(cls))
        return {
            "status":   "pass" if not unmapped else "warning",
            "warnings": [f"상위 클래스 없음: {u}" for u in unmapped]
        }

    def _check_duplicates(self, g) -> dict:
        seen, dupes = set(), []
        for s, p, o in g:
            key = (str(s), str(p), str(o))
            if key in seen:
                dupes.append(key)
            seen.add(key)
        return {"status": "pass" if not dupes else "warning",
                "duplicate_count": len(dupes),
                "warnings": [f"중복 트리플: {d}" for d in dupes[:5]]}

    def _check_consistency(self) -> dict:
        owl_path = Path("data/rdf/oral-history.owl")
        if not owl_path.exists():
            return {"status": "skip", "warnings": ["OWL 파일 없음"]}
        try:
            from owlready2 import get_ontology
            onto = get_ontology(f"file://{owl_path.resolve()}")
            onto.load()
            inconsistent = []
            # 간단 체크만 (HermiT은 무거우므로 경량 버전)
            return {"status": "pass", "errors": inconsistent}
        except Exception as e:
            return {"status": "warning", "warnings": [str(e)]}

        pkg_dir = self.publish_dir / f"v{config['version']}"
        pkg_dir.mkdir(parents=True, exist_ok=True)

                shutil.copy2(src, pkg_dir / fname)
                files_created.append(fname)

        # README 생성
        readme = self._generate_readme(config)
        (pkg_dir / "README.md").write_text(readme, encoding="utf-8")
        files_created.append("README.md")

        # CHANGELOG 생성/복사
        changelog_src = Path("CHANGELOG_ONTOLOGY.md")
        if changelog_src.exists():
            shutil.copy2(changelog_src, pkg_dir / "CHANGELOG.md")
        else:
            (pkg_dir / "CHANGELOG.md").write_text(
                f"# 온톨로지 변경 이력\n\n"
                f"## [{config['version']}] - "
                f"{datetime.now().strftime('%Y-%m-%d')}\n"
                f"### 최초 공표\n", encoding="utf-8")
        files_created.append("CHANGELOG.md")

        return {"package_dir": str(pkg_dir), "files": files_created}

    def preview_readme(self) -> dict:
        config = {"name": "한국 구술기록 온톨로지",
                  "version": "1.0.0", "uri": "", "license": "", "description": "", "author": ""}
        history_path = Path("data/publish_history.json")
        if history_path.exists():
            history = json.loads(history_path.read_text(encoding="utf-8"))
            if history:
                config.update(history[-1].get("config", {}))
        return {"readme": self._generate_readme(config)}

    def publish(self, config: dict) -> dict:
        result = self.create_package(config)
        pkg_dir = Path(result["package_dir"])
        output  = Path(config.get("output_path", ""))

        if output and output.exists():
            for f in pkg_dir.iterdir():
                shutil.copy2(f, output / f.name)
            result["published_to"] = str(output)

        # 이력 저장
        history = self._load_history()
        history.append({
            "version":    config["version"],
            "timestamp":  datetime.now().isoformat(),
            "config":     config,
            "files":      result["files"],
            "output":     config.get("output_path", "local"),
        })
        self.history_path.write_text(
            json.dumps(history, ensure_ascii=False, indent=2), encoding="utf-8")
        return {"status": "published", **result}

    def get_history(self) -> list:
        return self._load_history()

    def _generate_readme(self, config: dict) -> str:
        return f"""# {config['name']} v{config['version']}

> {config['description']}

## 네임스페이스

```turtle
@prefix ora: <{config['uri']}#> .
```

## 참조 온톨로지

- CIDOC-CRM 7.1.2
- RiC-O 1.1
- FOAF
- Schema.org

## 라이선스

[{config['license']}]({config['license']})

## 저자

{config['author']}

## 버전 이력

CHANGELOG.md 참조
"""

    def _load_history(self) -> list:
        if self.history_path.exists():
            return json.loads(self.history_path.read_text(encoding="utf-8"))
        return []
```

```python
# api/router_rdf_export.py
from fastapi import APIRouter, Query
from fastapi.responses import Response
from core.rdf_store import get_rdf_store

router = APIRouter(prefix="/rdf/export", tags=["RDF 내보내기"])

FORMATS = {
    "turtle":  ("text/turtle",          ".ttl"),
    "json-ld": ("application/ld+json",  ".jsonld"),
    "xml":     ("application/rdf+xml",  ".owl"),
}

@router.get("/ontology")
def export_ontology(
    format: str = Query("turtle", enum=list(FORMATS.keys()))
):
    store = get_rdf_store()
    media_type, ext = FORMATS[format]
    content = store.serialize(format=format)
    return Response(content=content, media_type=media_type,
        headers={"Content-Disposition":
                 f"attachment; filename=oral-history{ext}"})

@router.get("/triples")
def export_triples(
    format: str = Query("turtle", enum=list(FORMATS.keys())),
    selected_ids: str = Query(None)
):
    store = get_rdf_store()
    if selected_ids:
        ids = selected_ids.split(",")
        # 선택된 트리플만 필터링하는 쿼리 실행
        # (구체적 구현은 DB 스키마 확인 후)
    media_type, ext = FORMATS[format]
    content = store.serialize(format=format)
    return Response(content=content, media_type=media_type,
        headers={"Content-Disposition":
                 f"attachment; filename=triples{ext}"})
